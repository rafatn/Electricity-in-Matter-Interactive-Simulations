import numpy as np
from scipy import fft

class QuantumFrequencyEncoder:
    def __init__(self, base_freq=50):  # 50Hz רשת חשמל
        self.base_freq = base_freq
        
    def image_to_frequency(self, image):
        """המרת תמונה לתבנית תדר"""
        # דגימת פיקסלים
        pixels = self.sample_pixels(image)
        
        # המרה לתדרים
        frequency_pattern = self.pixels_to_frequencies(pixels)
        
        # הוספת תדר רשת כנושא
        encoded_signal = self.encode_on_grid_frequency(frequency_pattern)
        
        return encoded_signal
    
    def quantum_modulation(self, frequency_signal):
        """מודולציה קוונטית של תבנית התדר"""
        # יצירת מצב קוונטי
        quantum_state = self.create_quantum_state(frequency_signal)
        
        # שזירה עם תדר רשת
        entangled_state = self.entangle_with_grid(quantum_state)
        
        return entangled_state
    
    def conservation_check(self, input_energy, output_energy):
        """בדיקת חוק שימור האנרגיה"""
        energy_difference = abs(input_energy - output_energy)
        conservation_factor = 1 - (energy_difference / input_energy)
        
        return conservation_factor > 0.999  # דיוק 99.9%

# פונקציה להמרת תמונה לתבנית תדר
def image_to_frequency_spectrum(image):
    # המרת תמונה לגווני אפור
    gray_image = rgb_to_grayscale(image)
    
    # טרנספורם פורייה דו-ממדי
    fft_result = np.fft.fft2(gray_image)
    fft_shifted = np.fft.fftshift(fft_result)
    
    # חילוץ ספקטרום אמפליטודות
    magnitude_spectrum = np.log(np.abs(fft_shifted) + 1)
    
    # מיפוי לתדר רשת
    frequency_pattern = map_to_grid_frequency(magnitude_spectrum)
    
    return frequency_pattern

# חוק שימור: אנרגיה = אינטגרל על ספקטרום התדרים
def energy_conservation_check(original_image, frequency_signal):
    # אנרגיית התמונה המקורית
    image_energy = np.sum(original_image ** 2)
    
    # אנרגיית תבנית התדר
    frequency_energy = np.sum(np.abs(frequency_signal) ** 2)
    
    # בדיקת שימור
    conservation_ratio = frequency_energy / image_energy
    
    return abs(1 - conservation_ratio) < 0.01  # שגיאה של פחות מ-1%
