📡 סקירה כללית של המערכת

מערכת תקשורת היברידית פורצת דרך המשלבת תקשורת לוויינית ישירה עם אבטחה מבוססת תשתית פיזית - רשת החשמל. המערכת מספקת פתרון תקשורת מאובטח באזורים מרוחקים או במצבי חירום.

רעיון מרכזי: שימוש בתדר ובטביעת האצבע הייחודית של רשת החשמל כאמצעי אימות והצפנה, בשילוב עם תקשורת IPv6 דרך לוויין.

🔐 מכניזם אבטחה מבוסס תדר רשת חשמל

רעיון ה-Physical Unclonable Function (PUF):

כל רשת חשמל יוצרת "טביעת אצבע" ייחודית עקב:

  • רעש רקע והפרעות
  • הרמוניות בתדר הבסיס (50/60Hz)
  • איכות מתח וצריכת חשמל משתנה
  • מאפיינים ייחודיים של שנאים וכבלים

קוד דוגמה לחילוץ חתימה:

class PowerGridFingerprint: def __init__(self, sample_rate=4000): self.sample_rate = sample_rate def extract_grid_signature(self): # קריאת נתונים מ-ADC samples = self.read_adc_samples() # ניתוח FFT fft = np.fft.fft(samples) freqs = np.fft.fftfreq(len(samples), 1/self.sample_rate) # מציאת תדר בסיס והרמוניות base_freq = self.find_peak_frequency(freqs, fft) harmonics = self.extract_harmonics(base_freq, freqs, fft) # חישוב רעש רקע noise_floor = self.calculate_noise_floor(fft) # יצירת חתימה ייחודית signature = self.create_signature( base_freq, harmonics, noise_floor ) return signature def generate_crypto_key(self, signature): """המרת חתימה למפתח קריפטוגרפי""" from cryptography.hazmat.primitives.kdf.hkdf import HKDF from cryptography.hazmat.primitives import hashes hkdf = HKDF( algorithm=hashes.SHA256(), length=32, # מפתח 256-bit salt=signature[:16], info=b"power_grid_key" ) return hkdf.derive(signature)

דוגמת קוד להתחלת עבודה:

#!/usr/bin/env python3 """ PowerGrid Secure Satellite Communication - Proof of Concept """ import numpy as np from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes from cryptography.hazmat.primitives import hashes from cryptography.hazmat.primitives.kdf.hkdf import HKDF class SecureGridComm: def __init__(self): self.grid_key = None self.session_key = None def measure_grid_frequency(self, duration=0.1): """מדידת תדר רשת והפקת entropy""" # בדוגמה אמיתית: קריאה מ-ADC t = np.linspace(0, duration, 4000) # סימולציה של רשת חשמל עם רעש ייחודי base_freq = 50 # Hz noise = np.random.normal(0, 0.1, len(t)) # רעש ייחודי signal = np.sin(2 * np.pi * base_freq * t) + 0.1 * noise # ניתוח FFT fft = np.fft.fft(signal) freqs = np.fft.fftfreq(len(signal), t[1]-t[0]) # חילוץ חתימה ייחודית signature = self.extract_signature(freqs, fft) return signature def establish_secure_channel(self): """הקמת ערוץ תקשורת מאובטח""" # שלב 1: מדידת תדר רשת grid_signature = self.measure_grid_frequency() # שלב 2: הפקת מפתח מהחתימה self.grid_key = self.derive_key(grid_signature) # שלב 3: יצירת מפתח סשן self.session_key = self.generate_session_key() print(f"✅ ערוץ מאובטח הוקם בהצלחה") print(f" מפתח רשת: {self.grid_key[:16].hex()}...") print(f" מפתח סשן: {self.session_key[:16].hex()}...") return True def encrypt_message(self, message): """הצפנת הודעה עם מפתח הסשן""" cipher = Cipher( algorithms.AES(self.session_key), modes.GCM(np.random.bytes(12)) # IV אקראי ) encryptor = cipher.encryptor() encrypted = encryptor.update(message) + encryptor.finalize() return encrypted, encryptor.tag # שימוש לדוגמה if __name__ == "__main__": comm = SecureGridComm() if comm.establish_secure_channel(): msg = b"Hello from secure grid network!" encrypted, tag = comm.encrypt_message(msg) print(f"📨 הודעה הוצפנה: {encrypted.hex()[:32]}...")
